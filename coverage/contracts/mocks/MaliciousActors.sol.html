import {IERC721} from "@openzeppelin/contracts/token/ERC721/IERC721.sol";
import {IERC721Receiver} from "@openzeppelin/contracts/token/ERC721/IERC721Receiver.sol";
import {ERC721} from "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import {SimpleMarket} from "../SimpleMarket.sol";
 
/**
 * @title 测试用的恶意 / 异常合约
 *
 * 这些合约【仅用于单元测试】，绝不是要部署上线的东西。
 * 它们的作用是模拟攻击者与异常场景，用来验证我们的市场合约是否真的扛得住。
 *
 * 安全领域的一条基本原则：
 *   你不写攻击合约去打自己的合约，就永远不知道防线是否真的有效。
 *   「看起来对」和「被攻击后仍然对」是两回事。
 */
 
/**
 * @notice 恶意买家：收到 NFT 的瞬间尝试重入 buy()
 *
 * 攻击原理：
 *   市场用 safeTransferFrom 把 NFT 转给买家时，如果买家是合约，
 *   会回调它的 onERC721Received —— 这是一个免费的执行窗口。
 *   攻击者在窗口里再次调用 buy()，试图在挂单状态还没更新时重复购买
 *   （或者更狠的场景：把同一枚 NFT 反复卖给自己、反复套出 ETH）。
 *
 * 我们期望的结果：
 *   市场的两道防线（先改状态 + nonReentrant）会让第二次 buy 失败，
 *   reentrySucceeded 保持 false。
 */
contract MaliciousBuyer is IERC721Receiver {
    SimpleMarket public market;
    address public nft;
    uint256 public tokenId;
    uint256 public price;
 
    /// @notice 重入尝试次数
    uint256 public reentryAttempts;
 
    /// @notice 重入是否成功 —— 测试断言它必须为 false
    bool public reentrySucceeded;
 
    /// @notice 发起攻击：调用市场的 buy
    function attack(address marketAddr, address nftAddr, uint256 tokenId_, uint256 price_) external payable {
        market = SimpleMarket(marketAddr);
        nft = nftAddr;
        tokenId = tokenId_;
        price = price_;
        market.buy{value: price_}(nftAddr, tokenId_);
    }
 
    /**
     * @dev 收到 NFT 时的回调 —— 攻击窗口就在这里
     *
     * 用 try/catch 捕获失败：如果市场防护生效，重入的 buy 会 revert，
     * 被 catch 住后我们照常返回正确的 selector，让外层交易继续完成。
     * 这样测试才能走到最后去检查 reentrySucceeded == false。
     */
    function onERC721Received(
        address,
        address,
        uint256,
        bytes calldata
    ) external returns (bytes4) {
        reentryAttempts += 1;
        // 尝试再次购买同一枚 NFT。若防护有效，这里必然失败
        try market.buy{value: price}(nft, tokenId) {
            reentrySucceeded = true; // 如果走到这里，说明市场被打穿了
        } catch {
            // 预期路径：被 nonReentrant 或 NotListed 挡下
        }
        return IERC721Receiver.onERC721Received.selector;
    }
 
    /// @dev 能接收退款
    receive() external payable {}
}
 
/**
 * @notice 拒绝收款的卖家：用来验证「钱货两清」的原子性
 *
 * 场景：卖家是一个合约，且它的 receive 会 revert（可能是被恶意设计，
 *      也可能是多签钱包配置错误、fallback 太复杂导致 gas 不够等）。
 *
 * 我们期望的结果：
 *   市场给卖家打款失败 → 整笔交易 revert → NFT 也不转移、挂单仍有效。
 *   绝不能出现「NFT 已经给了买家，但卖家没收到钱」这种单方面损失。
 *   这就是原子性：要么全成功，要么全回滚。
 */
contract RejectingSeller is IERC721Receiver {
    /// @notice 是否拒绝接收 ETH
    bool public rejectEth;
 
    constructor(bool rejectEth_) {
        rejectEth = rejectEth_;
    }
 
    /// @dev 挂单：先授权市场，再挂单。必须由本合约持有 NFT 时调用
    function listOnMarket(address marketAddr, address nftAddr, uint256 tokenId_, uint256 price_) external {
        IERC721(nftAddr).approve(marketAddr, tokenId_);
        SimpleMarket(marketAddr).list(nftAddr, tokenId_, price_);
    }
 
    /// @dev 正常接收 NFT（否则挂单前的铸造都无法完成）
    function onERC721Received(
        address,
        address,
        uint256,
        bytes calldata
    ) external returns (bytes4) {
        return IERC721Receiver.onERC721Received.selector;
    }
 
    /// @dev 收 ETH 时按配置拒绝 —— 这里是触发回滚的关键
    receive() external payable {
        if (rejectEth) {
            revert("RejectingSeller: I refuse ETH");
        }
    }
}
 
/**
 * @notice 拒收退款的买家：能正常接收 NFT，但收不进任何 ETH
 *
 * 场景：买家多付了 ETH（前端估算偏差、抢单时多给），市场在成交末尾会把
 *      多出来的部分退回去。如果这个买家是个"只进不出"的合约 ——
 *      它的 receive 直接 revert —— 退款就会失败。
 *
 * 这个合约覆盖的是 buy() 里的最后一道分支：
 *      uint256 excess = msg.value - price;
 *      if (excess > 0) {
 *          (bool refundOk, ) = payable(msg.sender).call{value: excess}("");
 *          if (!refundOk) revert EthTransferFailed();   // ← 本合约要打的就是这里
 *      }
 *
 * 我们要验证两件事：
 *   1. 退款失败必须 revert EthTransferFailed（而不是默默吞掉多付的钱，
 *      也不是让买家白拿 NFT）；
 *   2. 整笔交易原子回滚 —— NFT 回到卖家手上、挂单依然有效、买家没收到 NFT。
 *
 * 注意它与 RejectingSeller 的区别：
 *   RejectingSeller 卡的是"给卖家打款"这一步（发生在转 NFT 之后），
 *   本合约卡的是"给买家退款"这一步（发生在最后）。
 *   两步都必须触发整体回滚，缺一不可测。
 *
 * 为什么它必须正确实现 onERC721Received？
 *   因为市场转 NFT 用的是 safeTransferFrom。如果本合约不返回正确的 selector，
 *   交易会在转 NFT 那一步就失败，根本走不到退款分支，测不到我们想测的东西。
 */
contract RefundRejectingBuyer is IERC721Receiver {
    /// @notice 本次攻击的目标市场
    SimpleMarket public market;
 
    /// @notice 目标 NFT 合约与编号（留档，便于在测试中回读）
    address public nft;
    uint256 public tokenId;
 
    /// @notice 是否真的收到过 NFT。整笔回滚后它必然是 false
    bool public receivedNft;
 
    /**
     * @notice 发起购买：把调用者给的 ETH 原样转发给 buy
     *
     * 调用方要带 value = price + 多付部分，且必须严格大于 price，
     * 否则 buy 里的 if (excess > 0) 走不进去，测不到退款分支。
     */
    function attack(address marketAddr, address nftAddr, uint256 tokenId_) external payable {
        market = SimpleMarket(marketAddr);
        nft = nftAddr;
        tokenId = tokenId_;
        market.buy{value: msg.value}(nftAddr, tokenId_);
    }
 
    /// @dev 正常接收 NFT（必须返回 selector，否则进不到退款那一步）
    function onERC721Received(
        address,
        address,
        uint256,
        bytes calldata
    ) external returns (bytes4) {
        receivedNft = true;
        return IERC721Receiver.onERC721Received.selector;
    }
 
    /// @dev 一分钱 ETH 都不收 —— 退款必然失败
    receive() external payable {
        revert("RefundRejectingBuyer: I refuse ETH");
    }
 
    /// @dev fallback 也堵死，避免有人用带数据的 call 绕过 receive
    fallback() external payable {
        revert("RefundRejectingBuyer: I refuse ETH");
    }
}
 
/**
 * @notice 拒收 ETH 的收款地址：专门用来卡 withdrawFees 的打款分支
 *
 * 场景：owner 调用 withdrawFees(to) 提取手续费，但 to 是一个拒绝收款的合约。
 *      市场里那行代码是：

        gotEth = true; // 这一行会随下面的 revert 一起回滚，实际永远是 false
